cmake_minimum_required(VERSION 3.22)
project(litz_package NONE)
if(NOT DEFINED LITZ_BINARY OR NOT DEFINED LITZ_MODEL)
  message(FATAL_ERROR "Set LITZ_BINARY to a release litzd and LITZ_MODEL to DeepFilterNet3_onnx.tar.gz")
endif()
set(LITZ_VERSION "0.1.0" CACHE STRING "Package version")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(BINDIR "${CPACK_PACKAGING_INSTALL_PREFIX}/bin")
configure_file(litzd.service.in litzd.service @ONLY)
install(PROGRAMS "${LITZ_BINARY}" DESTINATION bin)
install(FILES "${LITZ_MODEL}" DESTINATION share/litz)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/litzd.service" DESTINATION lib/systemd/user)
install(FILES ../../LICENSE ../../THIRD_PARTY_NOTICES DESTINATION share/licenses/litz)
set(CPACK_PACKAGE_NAME litz-audio)
set(CPACK_PACKAGE_VERSION "${LITZ_VERSION}")
set(CPACK_PACKAGE_CONTACT "Litz Audio contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Use your Android phone as a microphone and listen to your computer")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libpipewire-0.3-0 (>= 1.0)")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RPM_PACKAGE_REQUIRES "pipewire-libs >= 1.0")
set(CPACK_RPM_PACKAGE_LICENSE "GPL-3.0-or-later")
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
include(CPack)
